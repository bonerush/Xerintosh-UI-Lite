Per-CPU 数据结构

/* kern_cpu.h */ typedef struct kern_per_cpu { uint8_t cpu_id; kern_task_t *current_task; // 当前运行的任务 kern_task_t *idle_task; // idle 任务 (per-CPU) kern_runq_t runq; // 本核就绪队列 kern_ctx_t idle_ctx; // idle 任务上下文 → 调度循环 uint64_t sched_ticks; // 本核 tick 计数 volatile bool need_resched; // 需要重新调度 volatile bool ipi_pending; // 核间中断标志 uint8_t task_count; // 本核任务数 uint8_t reserved[2]; // cache line 对齐 } kern_per_cpu_t; extern kern_per_cpu_t g_per_cpu[KERN_MAX_CPUS]; /* 便捷宏：直接访问当前 CPU 的字段 */ #define KERN_THIS_CPU kern_cpu_id() #define g_current_task (g_per_cpu[KERN_THIS_CPU].current_task) #define g_need_resched (g_per_cpu[KERN_THIS_CPU].need_resched)

启动序列

Core 0 (PRO_CPU) Core 1 (APP_CPU) ───────────── ───────────── app_main() ├── kern_init() [复位状态，等待] │ ├── kern_cpu_init() │ ├── kern_sched_init() │ ├── 启动 CCOMPARE0 ISR │ ├── 创建 idle_0 │ └── kern_smp_start_core1()─→ esp_rom_aps_digi_boot() │ kern_smp_entry_core1() │ ├── kern_cpu_init(1) │ ├── 启动 CCOMPARE0 ISR(1) │ ├── 创建 idle_1 │ └── kern_sched_loop(1) │ ├── kern_sched_loop(0) [调度循环运行中] │ ... │ ... │ 创建 app 任务 │ 创建 app 任务 │ (pin CPU 0 或负载均衡) │ (pin CPU 1 或负载均衡) 两个核心的调度循环完全独立、并行运行。

核间中断 IPI

/* kern_smp.c — 使用 Xtensa 软件中断实现 IPI */ /* 通过设置目标核的 interrupt 寄存器触发软件中断 */ void kern_smp_send_ipi(uint8_t target_cpu) { if (target_cpu == 0) asm volatile("wsr.intset 0x01"); // 触发 Core 0 软件中断 else asm volatile("wsr.intset 0x02"); // 触发 Core 1 软件中断 } /* IPI ISR — 最小化处理 */ void kern_ipi_isr(void *arg) { uint8_t cpu = KERN_THIS_CPU; g_per_cpu[cpu].ipi_pending = true; g_per_cpu[cpu].need_resched = true; // 触发重调度 // 清除软件中断 if (cpu == 0) asm volatile("wsr.intclear 0x01"); else asm volatile("wsr.intclear 0x02"); }

IPI 使用场景

场景触发
kern_task_kill(pid) — 目标任务在另一核运行标记任务为 ZOMBIE，发 IPI 强制 target_cpu 重调度
kern_runq_enqueue 到另一核入队后若目标核空闲，发 IPI 唤醒
kern_task_set_affinity — 迁移任务从当前核 dequeue → 目标核 enqueue → 发 IPI
任务创建时负载均衡选任务较少的核创建，不发 IPI（等待目标核自然调度）

任务亲和性

/* kern_task.h 新增字段 */ typedef struct kern_task { // ... 现有字段 ... uint8_t cpu_id; // 当前所在核 (0/1, KERN_CPU_ANY=自动) uint8_t affinity_mask; // 允许运行的核位掩码 (bit0=Core0, bit1=Core1) // ... } kern_task_t; /* 默认: affinity_mask = 0x03 (双核) + cpu_id = KERN_CPU_ANY (自动) */
与现有 kern_smp 的关键区别：
• 当前：通过 xTaskCreatePinnedToCore 固定核心，无运行时迁移
• 新设计：affinity_mask 控制允许的核心集合，cpu_id 可在运行时变更
• 当前：全局任务链表受 spinlock 保护 → 跨核竞争
• 新设计：每核独立 runqueue，入队/dequeue 无锁，仅跨核操作需要 IPI
