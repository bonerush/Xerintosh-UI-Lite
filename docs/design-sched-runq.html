kern_runq — Per-CPU 就绪队列

/* kern_runq.h */ #define KERN_PRIO_MAX 32 // 0=最高, 31=最低(仅 idle) #define KERN_PRIO_IDLE 31 #define KERN_PRIO_DEFAULT 16 #define KERN_TIME_SLICE_MS 10 // 默认时间片 typedef struct kern_runq { kern_task_t *queues[KERN_PRIO_MAX]; // 每优先级 FIFO 头 kern_task_t *queue_tails[KERN_PRIO_MAX]; // 每优先级 FIFO 尾 (O(1) append) uint32_t bitmap; // bit[i]=1 表示优先级 i 有任务 uint8_t count; // 队列中任务总数 } kern_runq_t; /* 操作 — 全部 O(1) */ void kern_runq_enqueue(kern_runq_t *rq, kern_task_t *t); kern_task_t* kern_runq_dequeue(kern_runq_t *rq); // 取最高优先级 kern_task_t* kern_runq_steal(kern_runq_t *rq); // 偷取最低优先级(work-steal 预留)

Runqueue 结构示意

bitmap[31:0] 0x0000C010 ┌─┴───────────────┐ 优先级 0: │ NULL │ (空) 优先级 1: │ NULL │ (空) ... 优先级 14: │ ⋯→ task_A → task_B → NULL ← 2 个任务，FIFO 优先级 15: │ ⋯→ task_C → NULL ← 1 个任务 优先级 16: │ NULL │ (空) ... 优先级 31: │ idle 任务(常驻) │ pick_next() = queues[__builtin_ctz(bitmap)] ← O(1) 取最高优先级头部

kern_sched — 调度器核心循环

/* 每个核心的调度循环（kern_sched.c） */ void kern_sched_loop(void *arg) { uint8_t cpu = (uint8_t)(uintptr_t)arg; while (1) { // 1. 回收僵尸任务 reap_zombies(cpu); // 2. 唤醒到期睡眠任务 kern_sleep_tick(cpu); // 3. 从就绪队列选择下一个任务 kern_task_t *next = kern_runq_pick_next(&g_per_cpu[cpu].runq); if (next != NULL) { // 上下文切换到任务 g_per_cpu[cpu].current = next; next->state = KERN_TASK_RUNNING; next->slice_remaining = KERN_TIME_SLICE_MS; kern_ctx_switch(&g_per_cpu[cpu].idle_ctx, &next->ctx); // ─── 从任务切换回来时从这里继续 ─── // (任务调用了 yield/exit 或被抢占) } else { // 无就绪任务：空闲等待中断 kern_cpu_idle(cpu); } } }

抢占机制

CCOMPARE0 定时器 (每核 1kHz) │ ├── ISR: kern_timer_isr() │ g_per_cpu[cpu].sched_ticks++ │ if (--current->slice_remaining == 0 || need_resched_check) │ g_per_cpu[cpu].need_resched = true │ ▼ ISR 返回时 if (g_per_cpu[cpu].need_resched): kern_ctx_switch_isr(¤t->ctx, &idle_ctx) ─── 回到调度循环 ─── 当前任务 re-enqueue pick next → switch to next

优先级抢占场景

时机动作
任务创建 (prio > current)标记 need_resched，下次 tick 时抢占
中断唤醒高优先级任务ISR 返回时直接调度到高优先级
mutex 释放有等待者唤醒等待者，若其优先级更高则 yield
时间片耗尽重新入队到同优先级队尾，选择下一个
与当前实现的关键区别：
1. 当前：调度器在独立的 kern_sched_loop FreeRTOS 任务中运行，通过信号量令牌与任务切换
2. 新设计：调度循环运行在 idle 任务的上下文中，通过 kern_ctx_switch 与任务共享 CPU — 没有"调度器任务"，只有 idle 任务和用户任务轮流运行
3. 当前：pick_next 后调用 kern_port_switch_to(阻塞等待任务完成)→ 调度器被信号量阻塞
4. 新设计：kern_ctx_switch 直接跳转到任务，任务 yield 时跳回 idle 继续循环 — 无需信号量
