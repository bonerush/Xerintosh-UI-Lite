硬件定时器：ESP32 CCOMPARE0

/* kern_timer.c — 使用 Xtensa 内部 CCOMPARE0 定时器 */ #define TICK_HZ 1000 // 1kHz = 1ms tick #define TICK_US 1000 /* Xtensa 特殊寄存器 */ #define XT_CCOUNT 234 // 周期计数（自动递增，每 CPU 周期 +1） #define XT_CCOMPARE0 240 // 比较器 0：ccount == ccompare0 → 中断 void kern_timer_init_per_cpu(uint8_t cpu) { uint32_t ticks_per_ms = esp_rom_get_cpu_ticks_per_us() * 1000; // 设置下次中断 = 当前计数值 + ticks_per_ms uint32_t ccount; asm volatile("rsr.ccount %0" : "=r"(ccount)); asm volatile("wsr.ccompare0 %0" :: "r"(ccount + ticks_per_ms)); // 使能 CCOMPARE0 中断（level 1，低优先级） xt_set_interrupt_handler(ETS_CCOMPARE0_INUM, kern_timer_isr_per_cpu, ...); xt_ints_on(1 << ETS_CCOMPARE0_INUM); }
为什么选 CCOMPARE0 而非 GPTimer？
• CCOMPARE0 是 Xtensa 核心内部定时器，零外设依赖，不经过 APB 总线
• 直接读写特殊寄存器（rsr/wsr），延迟确定（~4 周期）
• 每个核心有独立的 CCOUNT/CCOMPARE 寄存器，天然 Per-CPU
• GPTimer 是 ESP-IDF 外设驱动，内部可能依赖 FreeRTOS

Timer ISR 流程

kern_timer_isr_per_cpu(cpu): │ ├── 1. 重装载 CCOMPARE0 │ ccount = rsr.ccount │ wsr.ccompare0 = ccount + ticks_per_ms │ ├── 2. 更新 tick 计数 │ g_per_cpu[cpu].sched_ticks++ │ ├── 3. 时间片递减 │ if (current != NULL && --current->slice_remaining == 0) │ g_per_cpu[cpu].need_resched = true │ ├── 4. 处理软件定时器（仅 Core 0） │ kern_soft_timer_tick() // 检查到期，回调 │ └── 5. ISR 返回 if (need_resched && 不是嵌套中断) kern_ctx_switch_isr(current.ctx, idle_ctx) ─── 跳到调度循环，重新选择任务 ───

软件定时器

/* 按到期时间排序的单向链表 */ typedef struct kern_soft_timer { uint64_t expire_tick; // 绝对到期 tick 数 void (*callback)(void *arg); void *arg; bool periodic; uint32_t period_ticks; // 如果是周期性的 struct kern_soft_timer *next; } kern_soft_timer_t; /* 每 tick 只检查链表头部（O(1) 均摊）*/ void kern_soft_timer_tick(void) { uint64_t now = kern_ticks(); while (g_timer_head && g_timer_head->expire_tick <= now) { kern_soft_timer_t *t = g_timer_head; g_timer_head = t->next; if (t->callback) t->callback(t->arg); if (t->periodic) kern_soft_timer_add(t); // 重新插入 } }
